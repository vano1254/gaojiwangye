一、创建对象的四种方式

        
// 1. 字面量：最常用
var wzy_student = { name: "王梓杨", age: 20, major: "数字媒体" };

// 2. 构造函数：首字母大写，用 new 调用
function wzy_Student(name, age) {
    this.name = name;
    this.age = age;
}
var wzy_s1 = new wzy_Student("王梓杨", 20);

// 3. Object.create()：指定原型，常用于继承
var wzy_base = { intro: function () { return this.name; } };
var wzy_s2 = Object.create(wzy_base);
wzy_s2.name = "王梓杨";

// 4. class：ES6 语法糖，本质还是原型
class wzy_Person {
    constructor(name) { this.name = name; }
    intro() { return "我是" + this.name; }
}
var wzy_s3 = new wzy_Person("王梓杨");

        

            四种写法造出来的都是普通对象，区别在于原型链挂在哪：字面量挂在 Object.prototype，
            Object.create() 挂在你指定的对象上，new 挂在构造函数的 prototype 上。
        

    


    
    

        
二、访问属性的四种写法

        

            
                

                    写法
                    示例
                    什么时候用
                

            
            
                

                    点号
                    wzy_student.name
                    属性名是固定的、合法的标识符
                

                

                    方括号
                    wzy_student["name"]
                    属性名存在变量里、含空格或连字符时
                

                

                    可选链
                    wzy_student.school?.name
                    不确定中间那层存不存在，避免报错
                

                

                    解构
                    var { name, age } = wzy_student;
                    一次取出多个属性，写起来更短
                

            
        

        
// 方括号的杀手锏：属性名可以是变量
var wzy_key = "major";
console.log(wzy_student[wzy_key]);     // "数字媒体" —— 点号做不到这一点

// 可选链：没有 school 也不会报错，只会返回 undefined
console.log(wzy_student.school?.name); // undefined（不加 ? 会直接抛 TypeError）

// 解构 + 默认值 + 改名
var { name: wzy_n, age = 18, city = "未知" } = wzy_student;

    


    
    

        
三、遍历对象的三种姿势

        

            
                

                    方法
                    拿到什么
                    适合做什么
                

            
            
                

                    for...in
                    键名（会连原型上的可枚举属性一起遍历）
                    简单遍历，但要配 hasOwnProperty 过滤
                

                

                    Object.keys()
                    键名数组
                    只想拿键，或者要 map / forEach
                

                

                    Object.values()
                    值数组
                    求和、找最大值这类只关心值的场景
                

                

                    Object.entries()
                    [键, 值] 二维数组
                    键值都要用，配 for...of 解构最顺手
                

            
        

        
for (var wzy_k in wzy_student) {
    if (wzy_student.hasOwnProperty(wzy_k)) {
        console.log(wzy_k, wzy_student[wzy_k]);
    }
}

Object.keys(wzy_student).forEach(function (wzy_k) {
    console.log(wzy_k + " = " + wzy_student[wzy_k]);
});

for (var [wzy_k2, wzy_v] of Object.entries(wzy_student)) {
    console.log(wzy_k2, wzy_v);
}

    


    
    

        
四、对象方法里的 this：箭头函数会翻车

        
var wzy_user = {
    name: "王梓杨",

    // 普通函数：this 指向「调用它的那个对象」
    sayHi: function () {
        return "你好，" + this.name;
    },

    // 箭头函数：this 继承外层作用域，这里外层是 window，没有 name
    sayHiBad: () => "你好，" + this.name
};

console.log(wzy_user.sayHi());      // "你好，王梓杨"
console.log(wzy_user.sayHiBad());   // "你好，undefined"

        

            结论：对象的方法用普通函数写。箭头函数没有自己的 this，
            它只会往外层找，放在对象字面量里基本都拿不到这个对象。
        

    


    
    

        
五、原型链：方法为什么不占自己的内存

        
function wzy_Student(name, age) {
    this.name = name;
    this.age = age;
}

// 方法挂在 prototype 上，所有实例共用同一份
wzy_Student.prototype.intro = function () {
    return "我是" + this.name + "，今年" + this.age + "岁";
};

var wzy_s = new wzy_Student("王梓杨", 20);

wzy_s.intro();                      // "我是王梓杨，今年20岁"
wzy_s.hasOwnProperty("name");       // true  —— 自己的属性
wzy_s.hasOwnProperty("intro");      // false —— 方法在原型上，不在自己身上
"intro" in wzy_s;                   // true  —— in 会顺着原型链找

Object.getPrototypeOf(wzy_s) === wzy_Student.prototype;   // true

        

            查找顺序：先找自己 → 找不到顺着 __proto__ 往上找 → 一直到 Object.prototype → 还没有就是 undefined。
            这就是原型链。
        

    


    
    

        
六、对象与 JSON 的互转

        
var wzy_json = JSON.stringify(wzy_student);          // 对象 → 字符串
var wzy_json2 = JSON.stringify(wzy_student, null, 2); // 带缩进，方便看
var wzy_back = JSON.parse(wzy_json2);                 // 字符串 → 对象

// 两个坑：
JSON.stringify({ f: function () {} });   // "{}"        —— 函数会被丢掉
JSON.stringify({ u: undefined });        // "{}"        —— undefined 也会被丢掉
JSON.stringify({ d: new Date() });       // 日期会变成字符串

// 深拷贝的常用写法（对象里没有函数 / undefined / 循环引用时够用）
var wzy_copy = JSON.parse(JSON.stringify(wzy_student));

    


    
    

        
七、要点说明

        
    
            
  1. 对象是键值对的集合，键一定是字符串（或 Symbol），写数字键也会被转成字符串。
    2. 
            
  2. obj.name 和 obj["name"] 等价，但方括号里的键可以是变量，点号不行。
    3. 
            
  3. delete 只能删自己的属性，删不掉原型上的；删完再访问得到 undefined。
    4. 
            
  4. Object.assign(target, source) 改的是第一个参数，返回的也是它 —— 不是新对象。
    5. 
            
  5. 展开 { ...obj } 是浅拷贝：只复制第一层，嵌套对象还是同一个引用。
    6. 
            
  6. in 会查原型链，hasOwnProperty 只查自己，这是两者最关键的差别。
    7. 
            
  7. 对象的方法别用箭头函数写，否则 this 拿不到当前对象。
    8. 
            
  8. 本页所有操作都在真实对象上跑，左边 JSON 和右边属性表会同步变化，多试几次就摸清了。
